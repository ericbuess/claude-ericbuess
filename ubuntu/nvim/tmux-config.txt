# ============================================================================
# OPTIONAL: Add these lines to your ~/.tmux.conf for better live updates
# ============================================================================
# This enhances the live-reload experience when using Neovim to watch
# files being edited by Claude Code in another tmux pane

# Enable focus events (CRITICAL for auto-reload)
set -g focus-events on

# Reduce escape time for better responsiveness
set -g escape-time 0

# Enable mouse support (optional but helpful)
set -g mouse on

# Better colors
set -g default-terminal "screen-256color"
set -ga terminal-overrides ",*256col*:Tc"

# Increase history limit
set -g history-limit 10000

# To apply these changes, either:
# 1. Restart tmux, or
# 2. Run: tmux source-file ~/.tmux.conf
